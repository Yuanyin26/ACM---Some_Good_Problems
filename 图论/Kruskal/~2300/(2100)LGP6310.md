## P6310 「Wdsr-1」仓库建设

[P6310 「Wdsr-1」仓库建设 - 洛谷](https://www.luogu.com.cn/problem/P6310)

## 思路

对于一个起点仓库 $u$，只能走权值不比 $x_u$ 大的边。也就是说，对于 $u \rightarrow v$ 的一条合法路径，满足路径中边权的最大值小于等于 $x_u$。那么我们建立一个最小生成树， 对于任意两个 $u,v$ 而言，在这个树的简单路径一定是其最优的路径选择（也就是最大值最小）。

我们建立一颗 $kruskal$ 重构树，将连边的边权作为中间点的点权。对于 $u$ ，找到其最后一个点权不大于 $x_u$ 的祖先 $F$,那么以 $F$  为根的子树中所含的叶子集合 $V$, $u$ 都可以到达。

那么我们可以倍增找到每个叶子节点对应的 $F$ 并标记，然后从根节点往下 $dfs$ 。如果当前搜到的节点被标记，那就只需要在标记其的位置建立粮仓，就可以运送到整个子树了；这样我们统计第一个答案。

对于一个被占领的节点，如果被占领不影响初始答案被标记的情况，则答案不变；否则我们从其标记的节点开始往下 $dfs$，做相同的搜索即可。如果搜到了被占领的节点，那么该节点就无法被运输到，输出-1；否则，将额外需要建立的粮仓与出答案相加即可。

## 代码

```c++
#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using i64 = long long;
using ull = unsigned long long;
using i128 = __int128_t;
using pii = std::pair<int,int>;
constexpr int N = 1e6 + 7;
constexpr int P = 998244353;
using ull = unsigned long long;
using ld = long double;
struct dsu
{
    std::vector<int>fa;
    int n;
    dsu(int len)
    {
        n = len;
        fa.resize(n + 1);
        for(int i = 1;i <= n;i++)fa[i] = i;
    }
    int find(int x)
    {
        if(x == fa[x])return x;
        return fa[x] = find(fa[x]);
    }
    void merge(int a,int b)
    {
        a = find(a);b = find(b);
        if(a > b )std::swap(a,b);
        fa[a] = b;
    }
};
void solve()
{
    int n,m;
    std::cin >> n >> m;
    std::vector<std::array<int,3>>edge(m);
    for(int i = 0;i < m;i++)
    {
        int u,v,w;
        std::cin >> u >> v >> w;
        edge[i] = {w,u,v};
    }
    std::sort(edge.begin(),edge.end());
    dsu ds = dsu(2 * n);
    std::vector g(2 * n,std::vector<int>());
    std::vector<int>W(2 * n);
    int tot = n;
    for(int i = 0;i < m;i++)
    {
        auto [w,u,v] = edge[i];
        if(ds.find(u) == ds.find(v))continue;
        u = ds.find(u),v = ds.find(v);
        ds.fa[u] = ++tot;
        ds.fa[v] = tot;
        g[tot].push_back(u);
        g[tot].push_back(v);
        W[tot] = w;
    }
    for(int i = 1;i <= n;i++)std::cin >> W[i];
    std::vector fa(2 * n,std::vector<int>(21));
    std::vector<int>cnt(2 * n),pos(2 * n),mx(n + 1);
    auto dfs = [&](auto &&self,int u,int par) -> void
    {
        fa[u][0] = par;
        for(int j = 1;fa[fa[u][j - 1]][j - 1];j++)
            fa[u][j] = fa[fa[u][j - 1]][j - 1];
        int now = u;
        for(int j = 20;j >= 0;j--)
            if(fa[now][j] && W[u] >= W[fa[now][j]])now = fa[now][j];
        if(u <= n)
        {
            cnt[now]++;
            pos[now] = u;
            mx[u] = now;
        }
        for(int v : g[u])
        {
            if(v == par)continue;
            self(self,v,u);
        }
    };
    dfs(dfs,2 * n - 1,0);
    int ans = 0;
    std::vector<bool>vis(n + 1);
    auto dfs1 = [&](auto &&self,int u,int par) -> void
    {
        if(cnt[u])
        {
            ans++;
            vis[pos[u]] = 1;
            return;
        }
        for(int v : g[u])
        {
            if(v == par)continue;
            self(self,v,u);
        }
    };
    dfs1(dfs1,2 * n - 1,0);
    std::cout << ans << " ";
    auto dfs2 = [&](auto &&self,int u,int par,int ex) -> int
    {
        if(u == ex)return -1;
        if(cnt[u])return 1;
        int sum = 0;
        for(int v : g[u])
        {
            if(v == par)continue;
            int tmp = self(self,v,u,ex);
            if(tmp == -1)return -1;
            sum += tmp;
        }
        return sum;
    };
    for(int i = 1;i <= n;i++)
    {
        if(!vis[i])std::cout << ans << " \n"[i == n];
        else
        {
            cnt[mx[i]]--;
            int tmp = dfs2(dfs2,mx[i],fa[mx[i]][0],i);
            if(tmp == -1)std::cout << -1 << " \n"[i == n];
            else std::cout << ans + tmp - 1 << " \n"[i == n];
        }
    }
}

signed main()
{
    // init();
    IOS;
    int T = 1;;//std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
```
