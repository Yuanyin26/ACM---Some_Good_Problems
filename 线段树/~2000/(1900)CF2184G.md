# G. Nastiness of Segments

[Problem - G - Codeforces](https://codeforces.com/contest/2184/problem/G)

## 思路

注意到等式两边的值，一个是**非递增**的，一个是**递增**的

所以这个值只有可能为 $1$ 或 $0$

直接二分这个位置即可

## 代码

```c++
#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using ull = unsigned long long;
using pii = std::pair<int,int>;
const int N = 55;
const int M = 1e6 + 7;
const int mod = 998244353;
struct seg
{
    struct node
    {
        int l;
        int r;
        int mn;
    };
    std::vector<node>t;
    std::vector<int>a;
    int n;
    void push_up(int p)
    {
        t[p].mn = std::min(t[p << 1].mn,t[p << 1 | 1].mn);
    }
    void build(int p,int l,int r)
    {
        t[p] = {l,r,inf};
        if(l == r)
        {
            t[p].mn = a[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(p << 1,l,mid);
        build(p << 1 | 1,mid + 1,r);
        push_up(p);
    }
    seg(int len,std::vector<int> &arr)
    {
        n = len;
        t.resize((len << 2) + 1);
        a = arr;
        build(1,1,n);
    }
    void update(int p,int pos,int k)
    {
        if(t[p].l == t[p].r && t[p].l == pos)
        {
            t[p].mn = k;
            return;
        }
        int mid = (t[p].l + t[p].r) >> 1;
        if(pos <= mid)update(p << 1,pos,k);
        else update(p << 1 | 1,pos,k);
        push_up(p);
    }
    int query(int p,int nl,int nr)
    {
        if(nl <= t[p].l && t[p].r <= nr)
            return t[p].mn;
        int mid = (t[p].l + t[p].r) >> 1,res = inf;
        if(nl <= mid)res = std::min(res,query(p << 1,nl,nr));
        if(nr > mid)res = std::min(res,query(p << 1 | 1,nl,nr));
        return res;
    }
};
void solve()
{
    int n,q;
    std::cin >> n >> q;
    std::vector<int>arr(n + 1);
    for(int i = 1;i <= n;i++)std::cin >> arr[i];
    seg t = seg(n,arr);
    while(q--)
    {
        int op;std::cin >> op;
        if(op == 1)
        {
            int id,x;
            std::cin >> id >> x;
            t.update(1,id,x);
        }
        else
        {
            int l,r;
            std::cin >> l >> r;
            int L = l,R = r + 1;
            while(L + 1 != R)
            {
                int mid = (L + R) >> 1;
                int d = mid - l;
                if(t.query(1,l,mid) >= d)L = mid;
                else R = mid;
            }
            std::cout << ((L - l) == t.query(1,l,L)) << std::endl;
        }
    }
}

signed main()
{
    // init();
    IOS;
    int T = 1;std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
```
