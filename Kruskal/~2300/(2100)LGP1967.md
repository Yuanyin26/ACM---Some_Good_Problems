## P1967 [NOIP 2013 提高组] 货车运输

[P1967 [NOIP 2013 提高组\] 货车运输 - 洛谷](https://www.luogu.com.cn/problem/P1967)

## 思路

以最大生成树的方式建立一颗 $kruskal$ 重构树，$LCA(u,v)$ 的点权就是最终的答案。

## 代码

```c++
#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using i64 = long long;
using ull = unsigned long long;
using i128 = __int128_t;
using pii = std::pair<int,int>;
constexpr int N = 1e6 + 7;
constexpr int P = 998244353;
using ull = unsigned long long;
using ld = long double;

struct dsu
{
    std::vector<int>fa;
    int n;
    dsu(int len)
    {
        n = len;
        fa.resize(n + 1);
        for(int i = 1;i <= n;i++)fa[i] = i;
    }
    int find(int x)
    {
        if(x == fa[x])return x;
        return fa[x] = find(fa[x]);
    }
    void merge(int a,int b)
    {
        fa[find(a)] = find(b);
    }
};
void solve()
{
    int n,m;
    std::cin >> n >> m;
    std::vector<std::array<int,3>>edge(m);
    for(int i = 0;i < m;i++)
    {
        int u,v,w;
        std::cin >> u >> v >> w;
        edge[i] = {-w,u,v};
    }
    std::sort(edge.begin(),edge.end());
    std::vector<int>W(2 * n + 1);
    int tot = n;
    dsu ds = dsu(2 * n);
    std::vector g(2 * n + 1,std::vector<int>());
    for(auto [w,u,v] : edge)
    {
        int pu = ds.find(u);
        int pv = ds.find(v);
        if(pu == pv)continue;
        ds.fa[pu] = ++tot;
        ds.fa[pv] = tot;
        g[tot].push_back(pu);
        g[tot].push_back(pv);
        W[tot] = -w;
    }
    tot++;
    for(int i = 1;i < tot;i++)
        if(ds.find(i) == i)g[tot].push_back(i);
    W[tot] = -1;
    std::vector fa(tot + 1,std::vector<int>(17));
    std::vector<int>dep(tot + 1);
    auto dfs = [&](auto &&self,int u,int par) -> void
    {
        fa[u][0] = par;
        dep[u] = dep[par] + 1;
        for(int j = 1;fa[fa[u][j - 1]][j - 1] && j < 17;j++)
            fa[u][j] = fa[fa[u][j - 1]][j - 1];
        for(int v : g[u])
        {
            if(v == par)continue;
            self(self,v,u);
        }
    };
    dfs(dfs,tot,0);
    auto lca = [&](int x,int y) -> int
    {
        if(dep[x] < dep[y])std::swap(x,y);
        int res = dep[x] - dep[y];
        for(int i = 0;res;i++)
        {
            if(res & 1)x = fa[x][i];
            res >>= 1;
        }
        if(x == y)return x;
        for(int i = 16;i >= 0;i--)
            if(fa[x][i] != fa[y][i])
            {
                x = fa[x][i];
                y = fa[y][i];
            }
        return fa[x][0];
    };
    int q;std::cin >> q;
    while(q--)
    {
        int x,y;
        std::cin >> x >> y;
        int lc = lca(x,y);
        std::cout << W[lc] << "\n";
    }

}

signed main()
{
    // init();
    IOS;
    int T = 1;;//std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
```
