## L. Labyrinth

[Problem - L - Codeforces](https://codeforces.com/contest/1578/problem/L)

## 思路

以最大生成树的方式建立 $kruskal$ 重构树，一个由边建立的点分了两个部分的点。

两个部分之间的点相互到达必定经过最小的这条边，因为我要吃完所有糖果、所以我先吃完一个子树的糖果，再去吃另外一边的糖果更优。

这样可以算一个整体的最大初始值，并且这个初始值一定能保证吃完子树外糖果前永远是有效的。

这样 $dfs$ 到叶子节点，然后取最大的 $max$ 即可。

## 代码

```c++
#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using i64 = long long;
using ull = unsigned long long;
using i128 = __int128_t;
using pii = std::pair<int,int>;
constexpr int N = 1e6 + 7;
constexpr int P = 998244353;
using ull = unsigned long long;
using ld = long double;
struct dsu
{
    std::vector<int>fa;
    int n;
    dsu(int len)
    {
        n = len;
        fa.resize(n + 1);
        for(int i = 1;i <= n;i++)fa[i] = i;
    }
    int find(int x)
    {
        if(x == fa[x])return x;
        return fa[x] = find(fa[x]);
    }
    void merge(int a,int b)
    {
        fa[find(a)] = find(b);
    }
};
void solve()
{
    int n,m;
    std::cin >> n >> m;
    std::vector<int>c(2 * n);
    for(int i = 1;i <= n;i++)std::cin >> c[i];
    std::vector<std::array<int,3>>edge(m);
    for(int i = 0;i < m;i++)
    {
        int u,v,w;
        std::cin >> u >> v >> w;
        edge[i] = {-w,u,v};
    }
    std::sort(edge.begin(),edge.end());
    std::vector g(2 * n ,std::vector<int>());
    std::vector<int>W(2 * n);
    dsu ds = dsu(2 * n);
    int tot = n;
    for(auto [w,u,v] : edge)
    {
        int pu = ds.find(u);
        int pv = ds.find(v);
        if(pu == pv)continue;
        ds.fa[pu] = ++tot;
        ds.fa[pv] = tot;
        W[tot] = -w;
        g[tot].push_back(pu);
        g[tot].push_back(pv);
    }
    std::vector<int>siz(2 * n);
    int ans = 0;
    auto dfs = [&](auto &&self,int u) -> void
    {
        siz[u] = c[u];
        for(int v : g[u])
        {
            self(self,v);
            siz[u] += siz[v];
        }
    };
    dfs(dfs,tot);
    auto dfs1 = [&](auto &&self,int u,int mx) -> void
    {
        for(int v : g[u])
        {
            int res = siz[tot] - siz[v];
            self(self,v,std::min(mx,W[u] - res));
        }
        if(g[u].empty())ans = std::max(ans,mx);
    };
    dfs1(dfs1,tot,INF);
    if(ans <= 0)ans = -1;
    std::cout << ans << "\n";
}

signed main()
{
    // init();
    IOS;
    int T = 1;//std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
```
