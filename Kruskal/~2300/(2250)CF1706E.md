## E. Qpwoeirut and Vertices

[Problem - 1706E - Codeforces](https://codeforces.com/problemset/problem/1706/E)

## 思路

以最小生成树的方式建立 $kruskal$ 重构树，任意两个节点的 $LCA$ 代表的边编号就是最小的最大编号。预处理相邻两个节点的 $LCA$，对于区间查询，就是这个区间预处理的 $LCA$ 的最大值，这是一个充要条件。

## 代码

```c++
#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using i64 = long long;
using ull = unsigned long long;
using i128 = __int128_t;
using pii = std::pair<int,int>;
constexpr int N = 1e6 + 7;
constexpr int P = 998244353;
using ull = unsigned long long;
using ld = long double;
struct dsu
{
    std::vector<int>fa;
    int n;
    dsu(int len)
    {
        n = len;
        fa.resize(n + 1);
        for(int i = 1;i <= n;i++)fa[i] = i;
    }
    int find(int x)
    {
        if(x == fa[x])return x;
        return fa[x] = find(fa[x]);
    }
    void merge(int a,int b)
    {
        fa[find(a)] = find(b);
    }
};
struct LCA
{
    int root,n;
    std::vector<std::vector<int>>fa;
    std::vector<int>dep;
    inline void init(int u,std::vector<std::vector<int>> &g)
    {
        for(int i = 1;i < 21;i++)
            fa[u][i] = fa[fa[u][i - 1]][i - 1];
        for(int v : g[u])
        {
            if(v == fa[u][0])continue;
            fa[v][0] = u;
            dep[v] = dep[u] + 1;
            init(v,g);
        }
    }
    LCA(int len,int rt,std::vector<std::vector<int>> &g)
    {
        n = len,root = rt;
        fa.resize(n + 1,std::vector<int>(21));//maxn = 1e6;
        dep.resize(n + 1);
        init(root,g);
    }
    int lca(int x,int y)
    {
        if(dep[x] < dep[y])std::swap(x,y);
        int res = dep[x] - dep[y],i = 0;
        while(res)
        {
            if(res & 1)x = fa[x][i];
            res >>= 1;
            i++;
        }
        if(x == y)return x;
        for(int i = 20;i >= 0;i--)
            if(fa[x][i] != fa[y][i])x = fa[x][i],y = fa[y][i];
        return fa[x][0];
    }

};
void solve()
{
    int n,m,q;
    std::cin >> n >> m >> q;
    std::vector<std::array<int,3>>edge(m);
    for(int i = 0;i < m;i++)
    {
        int u,v;
        std::cin >> u >> v;
        edge[i] = {i + 1,u,v};
    }
    std::vector g(2 * n ,std::vector<int>());
    dsu ds = dsu(2 * n);
    std::vector<int>W(2 * n);
    int tot = n;
    for(auto [w,u,v] : edge)
    {
        int pu = ds.find(u);
        int pv = ds.find(v);
        if(pu == pv)continue;
        ds.fa[pu] = ++tot;
        ds.fa[pv] = tot;
        W[tot] = w;
        g[tot].push_back(pu);
        g[tot].push_back(pv);
    }
    LCA lc = LCA(tot,tot,g);
    std::vector st(n,std::vector<int>(21));
    for(int i = 1;i < n;i++)
        st[i][0] = W[lc.lca(i,i + 1)];
    for(int j = 1;j < 21;j++)
        for(int i = 1;i + (1LL << j) - 1 < n;i++)
            st[i][j] = std::max(st[i][j - 1],st[i + (1LL << j - 1)][j - 1]);
    while(q--)
    {
        int l,r;std::cin >> l >> r;
        if(l == r)std::cout << "0" << " \n"[!q];
        else
        {
            r--;
            int k = std::__lg(r - l + 1);
            int mx = std::max(st[l][k],st[r - (1LL << k) + 1][k]);
            std::cout << mx << " \n"[!q];
        }

    }
}

signed main()
{
    // init();
    IOS;
    int T = 1;std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
```
