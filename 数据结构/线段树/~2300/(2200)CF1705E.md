# E. Mark and Professor Koro

[Problem - E - Codeforces](https://codeforces.com/contest/1705/problem/E)

## 思路

把 $$x$$ 看成权值 $$2^x$$。操作把两个 $$x$$ 合并成一个 $$x+1$$，因此

$$
\sum_i 2^{a_i}
$$

始终不变。不断执行合并，相当于把这个和做二进制进位，最终最大的数就是该二进制数最高的 $$1$$ 所在的位置。

一次修改 $$a_i:x\to y$$，等价于对这个超长二进制数执行

$$
S\leftarrow S-2^x+2^y.
$$

于是只需要动态维护二进制位。

加上 $$2^p$$ 时：找到从 $$p$$ 开始第一个 $$0$$ 的位置 $$q$$，翻转 $[p,q]$，减去 $$2^p$$ 则是找到第一个 $1$ 的位置。

线段树维护每段中 $$0/1$$ 的数量，即可在线段树上找到“第一个 $$0$$”或“第一个 $$1$$”。

## 代码

```c++
#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using i64 = long long;
using ull = unsigned long long;
using i128 = __int128_t;
using pii = std::pair<int,int>;
using ull = unsigned long long;
using ld = long double;
constexpr int N = 2e5 + 27;
constexpr int P = 998244353;
struct node
{
    int l;
    int r;
    int sum;
    int lz = 0;
}t[N << 2];
int a[N];
void push_up(int p)
{
    t[p].sum = (t[p << 1].sum + t[p << 1 | 1].sum);
}
void push_down(int p)
{
    if(t[p].lz)
    {
        t[p << 1].sum = t[p << 1].r - t[p << 1].l + 1 - t[p << 1].sum;
        t[p << 1].lz ^= 1;
        t[p << 1 | 1].sum = t[p << 1 | 1].r - t[p << 1 | 1].l + 1 - t[p << 1 | 1].sum;
        t[p << 1 | 1].lz ^= 1;
        t[p].lz = 0;
    }
}
void build(int p,int l,int r)
{
    t[p] = {l,r,0,0};
    if(l == r)
    {
        t[p].sum = a[l];
        return;
    }
    int mid = (l + r) >> 1;
    build(p << 1,l,mid);
    build(p << 1 | 1,mid + 1,r);
    push_up(p);
}
void update(int p,int nl,int nr)
{
    if(nl <= t[p].l && t[p].r <= nr)
    {
        t[p].sum = t[p].r - t[p].l + 1 - t[p].sum;
        t[p].lz ^= 1;
        return;
    }
    push_down(p);
    int mid = (t[p].l + t[p].r) >> 1;
    if(nl <= mid)update(p << 1,nl,nr);
    if(nr > mid)update(p << 1 | 1,nl,nr);
    push_up(p);
}
int querya(int p,int nl,int nr)// first 0
{
    if(nl <= t[p].l && t[p].r <= nr)
    {
        if(!t[p].sum)return t[p].l;
        else if(t[p].sum == t[p].r - t[p].l + 1)return inf;
        
        push_down(p);
        if(t[p << 1].sum != t[p << 1].r - t[p << 1].l + 1)return querya(p << 1,nl,nr);
        else return querya(p << 1 | 1,nl,nr);
    }
    push_down(p);
    int mid = (t[p].l + t[p].r) >> 1,ans = inf;
    if(nl <= mid)ans = std::min(ans,querya(p << 1,nl,nr));
    if(nr > mid)ans = std::min(ans,querya(p << 1 | 1,nl,nr));
    return ans;
}
int queryb(int p,int nl,int nr)//first 1
{
    if(nl <= t[p].l && t[p].r <= nr)
    {
        if(t[p].sum == t[p].r - t[p].l + 1)return t[p].l;
        else if(!t[p].sum)return inf;

        push_down(p);

        if(t[p << 1].sum)return queryb(p << 1,nl,nr);
        else return queryb(p << 1 | 1,nl,nr);
    }
    push_down(p);
    int mid = (t[p].l + t[p].r) >> 1,ans = inf;
    if(nl <= mid)ans = std::min(ans,queryb(p << 1,nl,nr));
    if(nr > mid)ans = std::min(ans,queryb(p << 1 | 1,nl,nr));
    return ans;
}
int query(int p)
{
    if(t[p].sum == t[p].r - t[p].l + 1)return t[p].r;
    push_down(p);
    if(t[p << 1 | 1].sum)return query(p << 1 | 1);
    else return query(p << 1);
}
void solve()
{
    int n,q;
    std::cin >> n >> q;
    std::vector<int>arr(n + 1);
    for(int i = 1;i <= n;i++)
    {
        std::cin >> arr[i];
        a[arr[i]]++;
    }
    for(int i = 1;i < N - 1;i++)
    {
        a[i + 1] += a[i] >> 1;
        a[i] %= 2;
    }
    build(1,1,N - 1);
    while(q--)
    {
        int k,l;
        std::cin >> k >> l;
        int r = queryb(1,arr[k],N - 1);
        update(1,arr[k],r);

        arr[k] = l;
        r = querya(1,arr[k],N - 1);
        update(1,arr[k],r);
        std::cout << query(1) << "\n";
    }


    
}
signed main()
{
    // init();
    IOS;
    int T = 1;//std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
```
