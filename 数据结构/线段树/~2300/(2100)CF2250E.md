# E. Double-Rift Dial

[Problem - E - Codeforces](https://codeforces.com/contest/2250/problem/E)

## 思路

枚举前缀不现实，我们可以枚举当前右端点，并维护所有到当前端点的前缀。对于每个起点，如果当前前缀的块大于 $2$，则不合法。每次我们放入一个新的端点，会发现其只和新点值的左边和右边是否存在有关。根据这个状态来判断不同前缀值得变化情况，使用线段树来区间更新，一旦某个起点的前缀不合法后，标记即可。

## 代码

```c++
#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using i64 = long long;
using ull = unsigned long long;
using i128 = __int128_t;
using pii = std::pair<int,int>;
constexpr int N = 1e6 + 7;
constexpr int P = 1e9 + 7;
using ull = unsigned long long;
using ld = long double;
struct segment
{
    struct node
    {
        int l;
        int r;
        int cnt;
        int lz;
    };
    std::vector<node>t;
    int n;
    void push_up(int p)
    {
        t[p].cnt = std::max(t[p << 1].cnt,t[p << 1 | 1].cnt);
    }
    void push_down(int p)
    {
        if(t[p].lz)
        {
            t[p << 1].cnt += t[p].lz;
            t[p << 1].lz += t[p].lz;
            t[p << 1 | 1].cnt += t[p].lz;
            t[p << 1 | 1].lz += t[p].lz;
            t[p].lz = 0;
        }
    }
    void build(int p,int l,int r)
    {
        t[p] = {l,r,0,0};
        if(l == r)return;
        int mid = (l + r) >> 1;
        build(p << 1,l,mid);
        build(p << 1 | 1,mid + 1,r);
    }
    segment(int len)
    {
        n = len;
        t.resize((n + 1 << 2) + 7);
        build(1,1,n);
    }
    void update(int p,int nl,int nr,int k) 
    {
        if(nl > nr)return;
        if(nl <= t[p].l && t[p].r <= nr)
        {
            t[p].cnt += k;
            t[p].lz += k;
            return;
        }
        push_down(p);
        int mid = (t[p].l + t[p].r) >> 1;
        if(nl <= mid)update(p << 1,nl,nr,k);
        if(nr > mid)update(p << 1 | 1,nl,nr,k);
        push_up(p);
    }
    int dfs(int p)
    {
        if(t[p].l == t[p].r)
        {
            if(t[p].cnt > 2)
            {
                t[p].cnt = -inf;
                return 1;
            }
        }
        push_down(p);
        int mid = (t[p].l + t[p].r) >> 1,res = 0;
        if(t[p << 1].cnt > 2)res += dfs(p << 1);
        if(t[p << 1 | 1].cnt > 2)res += dfs(p << 1 | 1);
        push_up(p);
        return res;
    }
};
void solve()
{
    int n;std::cin >> n;
    std::vector<int>p(n * 2 + 1),ind(n + 2);
    for(int i = 1;i <= n;i++)
    {
        std::cin >> p[i];
        p[i + n] = p[i];
    }
    segment seg = segment(n);
    int ans = n;
    for(int r = 1;r < 2 * n;r++)
    {
        int x = p[r];
        int L = std::max(1LL,r - n + 1);
        int R = std::min(n,r);
        int i = ind[x - 1] >= L ? ind[x - 1] : 0;
        int j = ind[x + 1] >= L ? ind[x + 1] : 0;
        i = std::min(i,R);
        j = std::min(j,R);
        if(i > j)std::swap(i,j);
        if(i)
        {
            seg.update(1,L,i,-1);
            seg.update(1,j + 1,R,1);
        }
        else if(j)seg.update(1,j + 1,R,1);
        else seg.update(1,L,R,1);
        ans -= seg.dfs(1);
        ind[x] = r;
    }
    std::cout << ans << "\n";
}
signed main()
{
    // init();
    IOS;
    int T = 1;std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
```
