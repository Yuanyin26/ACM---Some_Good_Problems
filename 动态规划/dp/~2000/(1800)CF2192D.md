# D. Cost of Tree

[Problem - 2192D - Codeforces](https://codeforces.com/problemset/problem/2192/D)

## 思路

首先对于一个节点 $u$ 的子树，其变化只在字数当中。

那么改变的情况只有两种：一种是儿子节点 $v$ 的子树内部的改变；一种是将一个儿子节点整个子树放到另外一个儿子子树中最深的叶子上。对于第二种情况、每个儿子只需要考虑转移到拥有最深叶子的儿子和拥有次深叶子的儿子（如果要转移的儿子子树是拥有最深叶子的儿子子树的话）即可。

## 代码

```c++
#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using ull = unsigned long long;
using pii = std::pair<int,int>;
const int N = 1e6 + 7;
const int M = 1e6 + 7;
const int mod = 998244353;
void solve()
{
    int n;
    std::cin >> n;
    std::vector<int>a(n + 1),dp(n + 1),siz(n + 1),dep(n + 1);
    std::vector<int>mx(n + 1),zmx(n + 1);
    std::vector<std::vector<int>>g(n + 1);
    for(int i = 1;i <= n;i++)std::cin >> a[i];
    for(int i = 1;i < n;i++)
    {
        int u,v;
        std::cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    auto dfs = [&](auto &&self,int u,int fa) -> void
    {
        dep[u] = dep[fa] + 1;
        siz[u] = a[u];
        mx[u] = dep[u];
        for(int v : g[u])
        {
            if(v == fa)continue;
            self(self,v,u);
            int tmp = mx[v];
            if(tmp >= mx[u])
            {
                zmx[u] = mx[u];
                mx[u] = tmp;
            }
            else if(tmp > zmx[u])zmx[u] = tmp;
            siz[u] += siz[v];
            dp[u] += dp[v] + siz[v];
        }
    };
    dep[0] = -1;
    dfs(dfs,1,0);
    std::vector<int>ans(n + 1);
    auto upd = [&](auto &&self,int u,int fa) -> void
    {
        ans[u] = dp[u];
        for(int v : g[u])
        {
            if(v == fa)continue;
            self(self,v,u);
            ans[u] = std::max(ans[u],dp[u] - dp[v] + ans[v]);
            if(mx[v] == mx[u])ans[u] = std::max(ans[u],dp[u] + siz[v] * (zmx[u] - dep[v] + 1));
            else ans[u] = std::max(ans[u],dp[u] + siz[v] * (mx[u] - dep[v] + 1));
        }
    };
    upd(upd,1,0);
    for(int i = 1;i <= n;i++)
        std::cout << ans[i] << " \n"[i == n];
}

signed main()
{
    // init();
    IOS;
    int T = 1;std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
```
