# E. Tom and Jerry

[Problem - E - Codeforces](https://codeforces.com/contest/2245/problem/E)

## 思路

考虑选择的简单路径上的节点度数的奇偶性。结论：对于当前未被选择的边组成的森林中，如果选择的路径上的节点度数全部为偶数，那此时操作的人必败。

简单证明：

当前节点度都为 $0$ 时，失败；

不都为 $0$ 时，无论下一步如何操作，操作完后可选择的节点一定有度数为奇数的情况，则选择这个度变成奇数的节点 $u$，并选择另外一个奇度节点 $v$，使得 $u,v$ 的简单路径上除了端点外的节点度数都为偶数。将选择的边删掉后，对于再下一步操作的选手而言，可选择的节点的度数又都变成了偶数。所以当前操作的选手必败，得证。

也就是说我们选择的初始路径要满足只有两个端点的度数为奇数，中间节点的度数都为偶数。

#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using i64 = long long;
using ull = unsigned long long;
using i128 = __int128_t;
using pii = std::pair<int,int>;
constexpr int N = 1e6 + 7;
constexpr int P = 1e9 + 7;
using ull = unsigned long long;
using ld = long double;
struct dsu
{
    std::vector<int>fa,siz,w;
    int n;
    dsu(int len)
    {
        n = len;
        fa.resize(n + 1);
        siz.assign(n + 1,1);
        w.resize(n + 1);
        for(int i = 1;i <= n;i++)fa[i] = i;
    }
    dsu(int len,std::vector<int> &W)
    {
        n = len;
        fa.resize(n + 1);
        siz.assign(n + 1,1);
        w.assign(W.begin(),W.end());
        for(int i = 1;i <= n;i++)fa[i] = i;
    }
    int find(int x)
    {
        if(x == fa[x])return x;
        return fa[x] = find(fa[x]);
    }
    void merge(int a,int b)
    {
        a = find(a);
        b = find(b);
        if(a == b)return;
        if(siz[a] < siz[b])std::swap(a,b);
        siz[a] += siz[b];
        w[a] += w[b];
        fa[b] = a;
    }
};
void solve()
{
    int n;
    std::cin >> n;
    dsu ds = dsu(n);
    std::vector<int>du(n + 1);
    std::vector<pii>edge(n - 1);
    for(int i = 0;i < n - 1;i++)
    {
        int u,v;
        std::cin >> u >> v;
        edge[i] = {u,v};
        du[u]++;
        du[v]++;
    }
    std::vector<pii>res;
    int ans = 0;
    for(auto [u,v] : edge)
    {
        if(du[u] & 1 && du[v] & 1)
            ans++;
        else if(du[u] & 1 || du[v] & 1)
            res.push_back({u,v});
        else ds.merge(u,v);
    }
    std::vector<int>cnt(n + 1);
    for(auto [u,v] : res)
    {
        int x = du[u] & 1 ? v : u;
        x = ds.find(x);
        cnt[x]++;
    }
    for(int i = 1;i <= n;i++)
        ans += cnt[i] * (cnt[i] - 1) / 2;
    std::cout << ans << "\n";
}
signed main()
{
    // init();
    IOS;
    int T = 1;std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
