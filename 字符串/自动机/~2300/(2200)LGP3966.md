## P3966 [TJOI2013] 单词

[P3966 [TJOI2013\] 单词 - 洛谷](https://www.luogu.com.cn/problem/P3966)

## 思路

建立 **AC** 自动机，在建立的同时对每个字符串前缀的节点计数。

那么对于每一个前缀，都可以通过 $fail$ 树跳父亲得到新的存在的后缀节点。

## 代码

```c++
#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using i64 = long long;
using ull = unsigned long long;
using i128 = __int128_t;
using pii = std::pair<int,int>;
constexpr int N = 1e6 + 7;
constexpr int P = 998244353;
using ull = unsigned long long;
using ld = long double;
struct AC
{
    struct node
    {
        pii son[26];
        int fail;
        int cnt;
        void init()
        {
            memset(son,0,sizeof(son));
            fail = cnt = 0;
        }
    }tr[N];
    struct edge
    {
        int nx;
        int to;
    }e[N << 1];int heads[N],cntqq,tot;
    void add(int u,int v)
    {
        e[++cntqq] = {heads[u],v};
        heads[u] = cntqq;
    }
    void init(int n = N - 1)
    {
        cntqq = tot = 0;
        memset(heads,0,sizeof(int) * (n + 1));
        tr[0].init();
    }
    void insert(std::string s)
    {
        int pos = 0,n = s.size();
        for(int i = 0;i < n;i++)
        {
            int &son = tr[pos].son[s[i] - 'a'].first;
            if(!son)son = ++tot,tr[tot].init();
            tr[pos].son[s[i] - 'a'].second = 1;
            pos = son;
        }
        tr[pos].cnt++;
    }
    void build()
    {
        std::queue<int>q;
        for(int i = 0;i < 26;i++)
        {
            if(tr[0].son[i].second)
            {
                int v = tr[0].son[i].first;
                q.push(v);
                add(0,v);
            }
        }
        while(!q.empty())
        {
            int u = q.front();q.pop();
            for(int i = 0;i < 26;i++)
            {
                if(tr[u].son[i].second)
                {
                    tr[tr[u].son[i].first].fail = tr[tr[u].fail].son[i].first;
                    add(tr[tr[u].fail].son[i].first,tr[u].son[i].first);
                    q.push(tr[u].son[i].first);
                }
                else tr[u].son[i].first = tr[tr[u].fail].son[i].first;
            }
        }
    }
    void dfs1(int u)
    {
        for(int i = heads[u];i;i = e[i].nx)
        {
            int v = e[i].to;
            dfs1(v);
            tr[u].cnt += tr[v].cnt;
        }
    }
    void dfs2(int u)
    {
        for(int i = 0;i < 26;i++)
        {
            if(tr[u].son[i].second)
            {
                int v = tr[u].son[i].first;
                dfs2(v);
                tr[u].cnt += tr[v].cnt;
            }
        }
    }
    
}ac;
void solve()
{
    int n;std::cin >> n;
    ac.init();
    std::vector<std::string>arr(n + 1);
    for(int i = 1;i <= n;i++)
    {
        std::cin >> arr[i];
        ac.insert(arr[i]);
    }
    ac.build();
    ac.dfs2(0);
    ac.dfs1(0);
    for(int i = 1;i <= n;i++)
    {
        int pos = 0;
        for(char c : arr[i])
            pos = ac.tr[pos].son[c - 'a'].first;
        std::cout << ac.tr[pos].cnt << "\n";
    }

}

signed main()
{
    // init();
    IOS;
    int T = 1;//std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
```
