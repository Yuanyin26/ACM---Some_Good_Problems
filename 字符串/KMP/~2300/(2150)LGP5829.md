## P5829 【模板】失配树

[P5829 【模板】失配树 - 洛谷](https://www.luogu.com.cn/problem/P5829)

## 思路

对于每个位置，先得到其最长 **border**，长度不同时、更长的 **border** 去找其自身的最长 **border**，直到两个长度相同就找到了。这个部分和 **LCA** 如出一辙， 所以建出失配树，然后倍增求公共祖先即可。

## 代码

```c++
#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using i64 = long long;
using ull = unsigned long long;
using i128 = __int128_t;
using pii = std::pair<int,int>;
constexpr int N = 1e6 + 7;
constexpr int P = 998244353;
using ull = unsigned long long;
using ld = long double;
struct LCA
{
    int root,n;
    std::vector<std::vector<int>>fa;
    std::vector<int>dep;
    inline void init(int u,std::vector<std::vector<int>> &g)
    {
        for(int i = 1;;i++)
        {
            if(fa[u][i - 1] == -1)break;
            if(fa[fa[u][i - 1]][i - 1] == -1)break;
            fa[u][i] = fa[fa[u][i - 1]][i - 1];
        }
        for(int v : g[u])
        {
            if(v == fa[u][0])continue;
            fa[v][0] = u;
            dep[v] = dep[u] + 1;
            init(v,g);
        }
    }
    LCA(int len,int rt,std::vector<std::vector<int>> &g)
    {
        n = len,root = rt;
        fa.resize(n + 1,std::vector<int>(21,-1));//maxn = 1e6;
        dep.resize(n + 1);
        init(root,g);
    }
    int lca(int x,int y)
    {
        if(dep[x] < dep[y])std::swap(x,y);
        int res = dep[x] - dep[y],i = 0;
        while(res)
        {
            if(res & 1)x = fa[x][i];
            res >>= 1;
            i++;
        }
        if(x == y)return x;
        for(int i = 20;i >= 0;i--)
            if(fa[x][i] != fa[y][i])x = fa[x][i],y = fa[y][i];
        return fa[x][0];
    }

};
void solve()
{
    std::string s;std::cin >> s;
    int n = s.size();
    std::vector<int>pi(n);
    std::vector<int>f(n,n);
    std::vector<std::vector<int>>g(n + 1);
    g[n].push_back(0);
    for(int i = 1;i < n;i++)
    {
        int pos = pi[i - 1];
        while(pos && s[pos] != s[i])pos = pi[pos - 1];
        if(s[pos] == s[i])pos++;
        pi[i] = pos;
        f[i] = pos - 1;
        if(f[i] == -1)f[i] = n;
        g[f[i]].push_back(i);
    }
    LCA lc = LCA(n,n,g);
    int m;std::cin >> m;
    while(m--)
    {
        int p,q;std::cin >> p >> q;
        p = pi[p - 1] - 1;
        q = pi[q - 1] - 1;
        if(p == -1 || q == -1)
        {
            std::cout << "0\n";
            continue;
        }
        int fa = lc.lca(p,q);
        if(fa == n)std::cout << "0\n";
        else std::cout << fa + 1 << "\n";

    }
}

signed main()
{
    // init();
    IOS;
    int T = 1;//std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
```
