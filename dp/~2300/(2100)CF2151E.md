# E. Limited Edition Shop

[Problem - E - Codeforces](https://codeforces.com/contest/2151/problem/E)

## 思路

对于 $a$ 的每个数（编号 $x$)，我们可以选择买或者不买，如果不买，就必须由 $b$ 来买走，也就是此时对于 $b$ 而言，至少 $1$ ~ $pb_x - 1$ 的商品已经被买走。我们 $dp$ 这个状态。

每次先更新不买的情况，将 $0$ ~ $pb_x$ 的最大值更新给 $pb_x$ ，表示不买这个商品能到达的最大值；

然后更新 $0$ ~ $pb_x - 1$ ，表示 $a$ 买走这个商品。  

## 代码

```c++
#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using i64 = long long;
using ull = unsigned long long;
using i128 = __int128_t;
using pii = std::pair<int,int>;
using ull = unsigned long long;
using ld = long double;
constexpr int N = 2e5 + 7;
constexpr int P = 998244353;
struct seg
{
    struct node
    {
        int l;
        int r;
        int mx;
        int lz;
    };
    std::vector<node>t;
    void push_up(int p)
    {
        t[p].mx = std::max(t[p << 1].mx,t[p << 1 | 1].mx);
    }
    void push_down(int p)
    {
        if(t[p].lz)
        {
            t[p << 1].mx += t[p].lz;
            t[p << 1].lz += t[p].lz;
            t[p << 1 | 1].mx += t[p].lz;
            t[p << 1 | 1].lz += t[p].lz;
            t[p].lz = 0;
        }
    }
    void build(int p,int l,int r)
    {
        t[p] = {l,r,0,0};
        if(l == r)
            return;
        int mid = (l + r) >> 1;
        build(p << 1,l,mid);
        build(p << 1 | 1,mid + 1,r);
        push_up(p);
    }
    seg(int n)
    {
        t.resize((n << 2) + 7);
        build(1,1,n);
    }
    void update(int p,int nl,int nr,int k)
    {
        if(nl > nr)return;
        if(nl <= t[p].l && t[p].r <= nr)
        {
            t[p].mx += k;
            t[p].lz += k;
            return;
        }
        push_down(p);
        int mid = (t[p].l + t[p].r) >> 1;
        if(nl <= mid)update(p << 1,nl,nr,k);
        if(nr > mid)update(p << 1 | 1,nl,nr,k);
        push_up(p);
    } 
    void update(int p,int pos,int k)
    {
        if(t[p].l == t[p].r && t[p].l == pos)
        {
            t[p].mx = k;
            return;
        }
        push_down(p);
        int mid = (t[p].l + t[p].r) >> 1;
        if(pos <= mid)update(p << 1,pos,k);
        else update(p << 1 | 1,pos,k);
        push_up(p);
    } 
    int query(int p,int nl,int nr)
    {
        if(nl > nr)return 0;
        if(nl <= t[p].l && t[p].r <= nr)
            return t[p].mx;
        push_down(p);
        int mid = (t[p].l + t[p].r) >> 1,res = -INF; 
        if(nl <= mid)res = std::max(res,query(p << 1,nl,nr));
        if(nr > mid)res = std::max(res,query(p << 1 | 1,nl,nr));
        return res;
    }
};

void solve()
{
    int n;std::cin >> n;
    std::vector<int>v(n + 1),a(n + 1),b(n +1);
    for(int i = 1;i <= n;i++)std::cin >> v[i];
    for(int i = 1;i <= n;i++)std::cin >> a[i];
    for(int i = 1;i <= n;i++)
    {
        int pos;std::cin >> pos;
        b[pos] = i;
    }
    int sum = 0;
    seg ti = seg(n);
    for(int i = 1;i <= n;i++)
    {
        int ind = b[a[i]];
        int mx = std::max(sum,ti.query(1,1,ind));
        ti.update(1,ind,mx);//不吃，取个max
        ti.update(1,1,ind - 1,v[a[i]]);//吃掉
        sum += v[a[i]];
    }
    std::cout << std::max(sum,ti.t[1].mx) << "\n";
}
signed main()
{
    // init();
    IOS;
    int T = 1;std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
```
