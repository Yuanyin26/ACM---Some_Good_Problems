## P3715 [BJOI2017] 魔法咒语

[P3715 [BJOI2017\] 魔法咒语 - 洛谷](https://www.luogu.com.cn/problem/P3715)

## 思路

首先构建出忌讳词语的 **AC** 自动机。对于当前构筑的字符串，我们只需要保证最长匹配后缀中没有忌讳词语出现即可，也就是说后缀的状态可以压缩为 **AC** 自动机的节点状态。这样我们就可以考虑关于字符长度和后缀状态的转移方程了。

枚举每一个基础词汇 **s**，尝试拼接到当前字符的后面，如果当前后缀状态为 $u$，拼接后状态为 $v$，那就有一个转移 $dp[l + len_s][v] += dp[l][u]$，当然，从状态 $u$ 到状态 $v$ 的过程中，不允许出现忌讳词语的存在，否则无法更新。

那么当 $l$ 很小的时候，我们可以直接暴力的更新这个状态；当 $l > 100$ 时，注意到基础词汇长度小于等于 $2$，那么更新 $dp[l]$ 时，只需要 $dp[l - 1],dp[l - 2]$ 即可。同时转移方式是不会发生变化的，所以我们可以利用矩阵快速幂加速这个过程。

## 代码

```c++
#include<bits/stdc++.h>
#define IOS std::ios::sync_with_stdio(0);std::cin.tie(0);std::cout.tie(0);
#define inf 0x3f3f3f3f
#define INF 4557430888798830399
#define int long long
using i64 = long long;
using ull = unsigned long long;
using i128 = __int128_t;
using pii = std::pair<int,int>;
constexpr int N = 1e6 + 7;
constexpr int P = 1e9 + 7;
using ull = unsigned long long;
using ld = long double;
struct AC
{
    struct node
    {
        int son[26];
        int fail;
        bool fl;
        void init()
        {
            memset(son,0,sizeof(son));
            fail = fl = 0;
        }
    }tr[N];
    struct edge
    {
        int nx;
        int to;
    }e[N << 1];int heads[N],cntqq,tot;
    void add(int u,int v)
    {
        e[++cntqq] = {heads[u],v};
        heads[u] = cntqq;
    }
    void init(int n = N - 1)
    {
        cntqq = tot = 0;
        memset(heads,0,sizeof(int) * (n + 1));
        tr[0].init();
    }
    void insert(std::string s)
    {
        int pos = 0,n = s.size();
        for(int i = 0;i < n;i++)
        {
            int &son = tr[pos].son[s[i] - 'a'];
            if(!son)
                son = ++tot,tr[tot].init();
            pos = son;
        }
        tr[pos].fl = 1;
    }
    void build()
    {
        std::queue<int>q;
        for(int i = 0;i < 26;i++)
        {
            if(tr[0].son[i])
            {
                int v = tr[0].son[i];
                q.push(v);
                add(0,v);
            }
        }
        while(!q.empty())
        {
            int u = q.front();q.pop();
            for(int i = 0;i < 26;i++)
            {
                if(tr[u].son[i])
                {
                    tr[tr[u].son[i]].fail = tr[tr[u].fail].son[i];
                    add(tr[tr[u].fail].son[i],tr[u].son[i]);
                    q.push(tr[u].son[i]);
                }
                else tr[u].son[i] = tr[tr[u].fail].son[i];
            }
        }
    }
    void dfs1(int u)
    {
        for(int i = heads[u];i;i = e[i].nx)
        {
            int v = e[i].to;
            tr[v].fl = tr[v].fl | tr[u].fl;
            dfs1(v);
        }
    }
}ac;
struct mat
{
    std::vector<std::vector<int>>a;
    int n,m;
    mat(int x,int y)
    {
        n = x,m = y;
        a.resize(n + 1,std::vector<int>(m + 1));
    }
    int sizr()
    {
        return n;
    }
    int sizc()
    {
        return m;
    }
};
mat mul(mat &a,mat &b)
{
    int n = a.sizr();
    int m = b.sizc();
    mat c = mat(n,m);
    for(int i = 1;i <= n;i++)
        for(int j = 1;j <= m;j++)
            for(int k = 1;k <= a.sizc();k++)
            {
                c.a[i][j] += a.a[i][k] * b.a[k][j] % P;
                c.a[i][j] %= P;
            }
    return c;
}
mat qmi(mat a,int b)
{
    int len = a.sizr();
    mat res = mat(len,len);
    for(int i = 1;i <= len;i++)res.a[i][i] = 1;
    while(b)
    {
        if(b & 1)res = mul(res,a);
        a = mul(a,a);
        b >>= 1;
    }
    return res;
}
void solve()
{
    ac.init();
    int n,m,l;
    std::cin >> n >> m >> l;
    std::vector<std::string>arr(n);
    for(int i = 0;i < n;i++)std::cin >> arr[i];
    for(int i = 0;i < m;i++)
    {
        std::string s;std::cin >> s;
        ac.insert(s);
    }
    ac.build();
    ac.dfs1(0);
    if(l > 100)
    {
        int len = ac.tot + 1;
        mat st = mat(1,len * 2);
        mat a = mat(len * 2,len * 2);
        for(int i = 1;i <= len;i++)a.a[len + i][i] = 1;
        st.a[1][1] = 1;
        for(int u = 0;u <= ac.tot;u++)
        {
            if(ac.tr[u].fl)continue;
            for(int i = 0;i < n;i++)
            {
                int pos = u;
                bool fl = 0;
                for(char c : arr[i])
                {
                    pos = ac.tr[pos].son[c - 'a'];
                    if(ac.tr[pos].fl)fl = 1;
                }
                if(fl)continue;
                int x = u + 1,y = pos + 1;
                if(arr[i].size() == 1)
                {
                    a.a[len + x][len + y]++;
                    if(u == 0)st.a[1][len + y]++;
                }
                else
                    a.a[x][len + y]++;
            }
        }
        a = qmi(a,l);
        st = mul(st,a);
        int ans = 0;
        for(int i = 1;i <= len;i++)ans = (ans + st.a[1][i]) % P;
        std::cout << ans << "\n";
    }
    else
    {
        int len = ac.tot;
        std::vector dp (l + 1,std::vector<int>(len + 1));
        dp[0][0] = 1;
        for(int i = 0;i < l;i++)
        {
            for(int u = 0;u <= len;u++)
            {
                if(ac.tr[u].fl)continue;
                for(int j = 0;j < n;j++)
                {
                    int pos = u;
                    bool fl = 0;
                    for(char c : arr[j])
                    {
                        pos = ac.tr[pos].son[c - 'a'];
                        if(ac.tr[pos].fl)fl = 1;
                    }
                    if(fl)continue;
                    if(i + arr[j].size() <= l)
                    {
                        dp[i + arr[j].size()][pos] += dp[i][u];
                        dp[i + arr[j].size()][pos] %= P;
                    }
                }
            }
        }
        int ans = 0;
        for(int i = 0;i <= len;i++)ans = (ans + dp[l][i]) % P;
        std::cout << ans << "\n";
    }
}

signed main()
{
    // init();
    IOS;
    int T = 1;//std::cin >> T;
    while(T--)
    {
        solve();
    }
    return 0;
}
```
